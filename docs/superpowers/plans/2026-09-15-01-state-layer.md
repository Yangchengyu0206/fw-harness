# 狀態層（身分、票檔、索引、git hooks）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 實作 harness 的狀態層：以 git 身分為基礎的一票一檔票檔、狀態機與 DoD 檢查、動態 commit evidence 索引，以及 commit-msg / post-commit / post-merge hooks。

**Architecture:** 所有邏輯放在 `harnesslib` 套件（純函式 + 小型 I/O 包裝），CLI 腳本只做參數解析與輸出。git hooks 是 sh 薄包裝，呼叫 Python 腳本。所有檔案位於 `plugins/fw-c-harness/skills/fw-harness-init/template/`，之後由 `fw-harness-init` 原樣複製進 FW repo。

**Tech Stack:** Python ≥ 3.9 標準函式庫、git、POSIX sh（Git for Windows 內建）、pytest。

## Global Constraints

- Python 只用標準函式庫；最低版本 3.9（型別註記一律加 `from __future__ import annotations`）。
- 所有檔案讀寫使用 UTF-8；JSON 以 `ensure_ascii=False, indent=2` 輸出，結尾換行，換行字元 `\n`。
- CLI 啟動時呼叫 `cli.setup_io()`，避免 Windows cp950 主控台印中文或 emoji 時崩潰。
- 票號格式 `FW-NNNN`（至少 4 位數），檔案 `harness/tickets/FW-NNNN.json`。
- 狀態：`backlog`、`next`、`active`、`verifying`、`done`、`blocked`。
- evidence 種類：`check`、`review`、`hil`、`note`；commit evidence 不寫入票檔。
- 身分只來自 `git config user.name/email`，可由 `harness/people.json` 覆寫 slug / 顯示名稱。
- 使用者看得到的訊息一律繁體中文，錯誤訊息要寫明怎麼修。
- 測試指令：`py -3 -m pytest`（本機 `python` 是 Microsoft Store stub，不要用）。
- 本計畫所有路徑中 `TEMPLATE` = `plugins/fw-c-harness/skills/fw-harness-init/template`。

```python
def git(repo: Path, *args: str) -> str:
    proc = subprocess.run(
        ["git", *args], cwd=repo, capture_output=True, text=True, encoding="utf-8", errors="replace",
    )
    if proc.returncode != 0:
        raise AssertionError(f"git {' '.join(args)} 失敗：{proc.stderr}")

def commit(repo: Path, message: str, who=None) -> str:
    who = who or ALICE
    git(repo, "add", "-A")
    git(
        repo, "-c", f"user.name={who.name}", "-c", f"user.email={who.email}",
        "commit", "--allow-empty", "--no-verify", "-q", "-m", message,
    )
    return git(repo, "rev-parse", "--short", "HEAD").strip()


def init_repo(path: Path, who=None) -> Path:
    who = who or ALICE
    path.mkdir(parents=True, exist_ok=True)
    git(path, "init", "-q", "-b", "main")
    git(path, "config", "user.name", who.name)
    git(path, "config", "user.email", who.email)
    git(path, "config", "core.autocrlf", "false")
    (path / "harness" / "tickets").mkdir(parents=True, exist_ok=True)
    (path / "harness" / "tickets" / ".gitkeep").write_text("", encoding="utf-8")
    commit(path, "chore: init", who)
    return path


def make_origin(repo: Path, tmp_path: Path) -> Path:
    """為 repo 建立 bare origin 並推送 main。"""
    origin = tmp_path / "origin.git"
    git(tmp_path, "init", "-q", "--bare", "-b", "main", str(origin))
    git(repo, "remote", "add", "origin", str(origin))
    git(repo, "push", "-q", "origin", "main")
    git(repo, "fetch", "-q", "origin", "main")
    return origin


def clone_as(origin: Path, dest: Path, who) -> Path:

    git(dest, "config", "user.name", who.name)
    git(dest, "config", "user.email", who.email)
    git(dest, "config", "core.autocrlf", "false")
```

```python
"""UTF-8 JSON 讀寫。寫入採「暫存檔 + rename」，避免寫到一半損毀。"""
from __future__ import annotations

import json
import os
import tempfile
from pathlib import Path
from typing import Any

_MISSING = object()


def read_json(path: Path, default: Any = _MISSING) -> Any:
    path = Path(path)
    try:
        text = path.read_text(encoding="utf-8")
    except FileNotFoundError:
        if default is _MISSING:
            raise
        return default
    try:
        return json.loads(text)
    except json.JSONDecodeError as exc:
        raise ValueError(f"{path}: JSON 無法解析：{exc}") from exc


def dumps(data: Any) -> str:
    return json.dumps(data, ensure_ascii=False, indent=2) + "\n"


def write_json_atomic(path: Path, data: Any) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(prefix=f".{path.name}.", suffix=".tmp", dir=path.parent)
    try:
        with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as handle:
            handle.write(dumps(data))
        os.replace(tmp, path)
    except BaseException:
        if os.path.exists(tmp):
            os.unlink(tmp)
        raise
```

```python
"""git 指令的薄包裝。輸出一律以 UTF-8 解碼。"""
from __future__ import annotations

import subprocess
from pathlib import Path


class GitError(RuntimeError):
    """git 指令失敗。"""


def _run(repo: Path, args: list[str]) -> subprocess.CompletedProcess:
    return subprocess.run(
        ["git", *args], cwd=repo, capture_output=True, text=True, encoding="utf-8", errors="replace",
    )


def git(repo: Path, *args: str, check: bool = True) -> str:
    proc = _run(repo, list(args))
    if proc.returncode != 0:
        if check:
            raise GitError(f"git {' '.join(args)} 失敗：{proc.stderr.strip()}")
        return ""
    return proc.stdout


def ok(repo: Path, *args: str) -> bool:
    return _run(repo, list(args)).returncode == 0


def repo_root(start: Path) -> Path:
    return Path(git(start, "rev-parse", "--show-toplevel").strip())


def head(repo: Path) -> str:
    return git(repo, "rev-parse", "HEAD").strip()


def has_commits(repo: Path) -> bool:
    return ok(repo, "rev-parse", "--verify", "--quiet", "HEAD")


def ref_exists(repo: Path, ref: str) -> bool:
    return ok(repo, "rev-parse", "--verify", "--quiet", f"{ref}^{{commit}}")


def is_ancestor(repo: Path, commit: str, descendant: str) -> bool:
    if not commit:
        return False
    return ok(repo, "merge-base", "--is-ancestor", commit, descendant)


def list_dir_at(repo: Path, ref: str, directory: str) -> list[str]:
    out = git(repo, "ls-tree", "--name-only", f"{ref}:{directory}", check=False)
    return [line for line in out.splitlines() if line]


def show_at(repo: Path, ref: str, path: str) -> str | None:
    proc = _run(repo, ["show", f"{ref}:{path}"])
    return proc.stdout if proc.returncode == 0 else None


def fetch(repo: Path, main_ref: str) -> bool:
    """main_ref 形如 origin/main。沒有 remote 或離線時回傳 False。"""
    remote, _, branch = main_ref.partition("/")
    if not branch:
        return False
    return ok(repo, "fetch", "--quiet", remote, branch)
```

```python
"""協作身分：一律來自 git config，可由 harness/people.json 覆寫 slug 與顯示名稱。"""
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path

from . import gitutil, jsonio


class IdentityError(RuntimeError):
    """無法取得協作身分。"""


@dataclass(frozen=True)
class Identity:
    name: str
    email: str
    slug: str

    def as_ref(self) -> dict:
        return {"name": self.name, "email": self.email}


def slugify_email(email: str) -> str:
    local = email.split("@", 1)[0]
    if "+" in local:
        local = local.split("+", 1)[1]
    slug = re.sub(r"[^a-z0-9-]", "-", local.lower())
    slug = re.sub(r"-{2,}", "-", slug).strip("-")
    if not slug:
        raise IdentityError(f"無法從 email 產生檔名用的 slug：{email!r}；請在 harness/people.json 指定 slug")
    return slug


def current_identity(repo: Path) -> Identity:
    name = gitutil.git(repo, "config", "user.name", check=False).strip()
    email = gitutil.git(repo, "config", "user.email", check=False).strip()
    missing = [key for key, value in (("user.name", name), ("user.email", email)) if not value]
    if missing:
        fixes = "；".join(f'git config {key} "..."' for key in missing)
        raise IdentityError(f"缺少 git 身分（{', '.join(missing)}），所有 harness 紀錄都需要身分。請執行：{fixes}")
    people = jsonio.read_json(Path(repo) / "harness" / "people.json", default={})
    override = people.get(email) or people.get(email.lower()) or {}
```

```python
"""票檔模型、狀態機與團隊規則（spec §5）。票檔位於 harness/tickets/FW-NNNN.json，一票一檔。"""
from __future__ import annotations

import re
from pathlib import Path

from . import gitutil, jsonio
from .identity import Identity

TICKETS_DIR = Path("harness") / "tickets"
STATUSES = ("backlog", "next", "active", "verifying", "done", "blocked")
EVIDENCE_KINDS = ("check", "review", "hil", "note")
ID_RE = re.compile(r"^FW-(\d{4,})$")

STATUS_LEGEND = {
    "backlog": "尚未排入近期工作。",
    "next": "排定為接下來要做。",
    "active": "assignee 正在做；每人同時最多一張。",
    "verifying": "程式碼寫完、check 通過，等待 dod_pending（上板驗證、他人 review）。",
    "done": "DoD 全部滿足：hil evidence、非本人 review 且無未解 🔴。",
    "blocked": "因 blocked_reason 無法推進；解除後回到原狀態。",
}

EVIDENCE_FIELDS: dict = {
    "check": {"commit": str, "summary": str, "passed": bool},
    "review": {"ref": str, "open_critical": int},
    "hil": {"ref": str, "summary": str},
    "note": {"summary": str},
}


class TicketError(RuntimeError):
    """票檔操作失敗；訊息直接顯示給使用者。"""


def format_id(n: int) -> str:
    return f"FW-{n:04d}"


def parse_id(ticket_id: str) -> int:
    match = ID_RE.match(ticket_id or "")
    if not match:
        raise TicketError(f"票號格式錯誤：{ticket_id!r}（應為 FW-0001 形式）")
    return int(match.group(1))


def ticket_path(repo: Path, ticket_id: str) -> Path:
    parse_id(ticket_id)
    return Path(repo) / TICKETS_DIR / f"{ticket_id}.json"


def _is_person(value: object) -> bool:
    return isinstance(value, dict) and all(
        isinstance(value.get(key), str) and value.get(key) for key in ("name", "email")
    )


def _is_str_list(value: object) -> bool:
    return isinstance(value, list) and all(isinstance(item, str) for item in value)


def _matches(value: object, expected: type) -> bool:
    if expected is int:
        return isinstance(value, int) and not isinstance(value, bool)
    return isinstance(value, expected)


def validate_ticket(ticket: dict, filename: str | None = None) -> list[str]:
    errors: list[str] = []
    tid = ticket.get("id")
    if not isinstance(tid, str) or not ID_RE.match(tid):
        errors.append(f"id 格式錯誤：{tid!r}")
        tid = filename or "?"
    elif filename is not None and filename != f"{tid}.json":
        errors.append(f"{filename}：檔名與 id {tid} 不一致")

    def err(message: str) -> None:
        errors.append(f"{tid}：{message}")

    for key in ("title", "area", "user_visible_behavior", "notes", "created_at"):
        if not isinstance(ticket.get(key), str):
            err(f"{key} 必須是字串")
    status = ticket.get("status")
    if status not in STATUSES:
        err(f"status 必須是 {'/'.join(STATUSES)} 之一，目前是 {status!r}")
    if not _matches(ticket.get("priority"), int):
        err("priority 必須是整數")
    if not isinstance(ticket.get("requires_hil"), bool):
        err("requires_hil 必須是 true/false")
    for key in ("verification_steps", "dod_pending"):
        if not _is_str_list(ticket.get(key)):
            err(f"{key} 必須是字串陣列")
    if not _is_person(ticket.get("created_by")):
        err("created_by 必須包含 name 與 email")
    assignee = ticket.get("assignee")
    if assignee is not None and not _is_person(assignee):
        err("assignee 必須是 null 或包含 name 與 email")
    if status in ("active", "verifying", "done") and assignee is None:
        err(f"status 為 {status} 時必須有 assignee")
    reason = ticket.get("blocked_reason")
    if reason is not None and not isinstance(reason, str):
        err("blocked_reason 必須是 null 或字串")
    if status == "blocked" and not reason:
        err("status 為 blocked 時必須填 blocked_reason")

    evidence = ticket.get("evidence")
    if not isinstance(evidence, list):
        err("evidence 必須是陣列")
        evidence = []
    for index, item in enumerate(evidence):
        where = f"evidence[{index}]"
        if not isinstance(item, dict) or item.get("kind") not in EVIDENCE_FIELDS:
            err(f"{where} 的 kind 必須是 {'/'.join(EVIDENCE_KINDS)} 之一")
            continue
        if not _is_person(item.get("by")) or not isinstance(item.get("at"), str):
            err(f"{where} 必須有 by（name、email）與 at")
        for key, expected in EVIDENCE_FIELDS[item["kind"]].items():
            if not _matches(item.get(key), expected):
                err(f"{where}（{item['kind']}）缺少或型別錯誤：{key}")

    history = ticket.get("history")
    if not isinstance(history, list) or not history:
        err("history 必須是非空陣列")
        history = []
    for index, item in enumerate(history):
        valid = (
            isinstance(item, dict)
            and _is_person(item.get("by"))
            and isinstance(item.get("at"), str)
            and item.get("to") in STATUSES
            and (item.get("from") is None or item.get("from") in STATUSES)
        )
        if not valid:
            err(f"history[{index}] 必須有 by、at、from（null 或狀態）、to（狀態）")
    return errors


def load_ticket(repo: Path, ticket_id: str) -> dict:
    path = ticket_path(repo, ticket_id)
    if not path.exists():
        raise TicketError(f"找不到票 {ticket_id}（{path}）")
    return jsonio.read_json(path)


def save_ticket(repo: Path, ticket: dict) -> None:
    errors = validate_ticket(ticket)
    if errors:
        raise TicketError("票檔內容不合法：\n- " + "\n- ".join(errors))
    jsonio.write_json_atomic(ticket_path(repo, ticket["id"]), ticket)


def load_all(repo: Path) -> list[dict]:
    directory = Path(repo) / TICKETS_DIR
    if not directory.is_dir():
        return []
    return [jsonio.read_json(path) for path in sorted(directory.glob("FW-*.json"))]


def _numbers(names) -> set[int]:
    result = set()
    for name in names:
        match = ID_RE.match(Path(name).stem)
        if match and name.endswith(".json"):
            result.add(int(match.group(1)))
    return result


def allocate_id(repo: Path, main_ref: str) -> tuple[str, str | None]:
    local = _numbers(p.name for p in (Path(repo) / TICKETS_DIR).glob("FW-*.json"))
    warning = None
    remote: set[int] = set()
    if gitutil.ref_exists(repo, main_ref):
        remote = _numbers(gitutil.list_dir_at(repo, main_ref, TICKETS_DIR.as_posix()))
    else:
        warning = f"找不到 {main_ref}，只用本地票號分配；合併時可能撞號，合併前請執行 ticket.py collisions"
    return format_id(max(local | remote, default=0) + 1), warning


def create_ticket(
    repo: Path,
    who: Identity,
    *,
    title: str,
    area: str,
    now: str,
    main_ref: str,
    priority: int = 3,
    user_visible_behavior: str = "",
    verification_steps=(),
    requires_hil: bool = True,
) -> tuple[dict, str | None]:
    ticket_id, warning = allocate_id(repo, main_ref)
    if ticket_path(repo, ticket_id).exists():
        raise TicketError(f"{ticket_id} 已存在，請重試")
    ticket = {
        "id": ticket_id,
        "title": title,
        "area": area,
        "status": "backlog",
        "priority": priority,
        "requires_hil": requires_hil,
        "created_by": who.as_ref(),
        "created_at": now,
        "assignee": None,
        "user_visible_behavior": user_visible_behavior,
        "verification_steps": list(verification_steps),
        "dod_pending": [],
        "evidence": [],
        "history": [{"by": who.as_ref(), "at": now, "from": None, "to": "backlog", "note": "開票"}],
        "blocked_reason": None,
        "notes": "",
    }
    save_ticket(repo, ticket)
    return ticket, warning
```

```python
    for item in reversed(ticket["history"]):

    others = [

def _verifying_problems(repo: Path, ticket: dict) -> list[str]:
    head = gitutil.head(repo)
    for item in ticket["evidence"]:
        if item["kind"] == "check" and item.get("passed") is True and gitutil.is_ancestor(repo, item.get("commit", ""), head):
            return []
    return ["需要至少一筆通過的 check evidence，且其 commit 是 HEAD 或 HEAD 的祖先（執行 check.py --record <票號>）"]


def done_problems(ticket: dict) -> list[str]:
    problems = []
    if ticket["dod_pending"]:
        problems.append("dod_pending 尚未清空：" + "；".join(ticket["dod_pending"]))
    evidence = ticket["evidence"]
    if ticket["requires_hil"] and not any(item["kind"] == "hil" for item in evidence):
        problems.append("需要 hil evidence（上板驗證紀錄）")
    reviews = [item for item in evidence if item["kind"] == "review"]
    if not reviews:
        problems.append("需要 review evidence（由 assignee 以外的人執行 fw-c-review）")
    else:
        latest = reviews[-1]
        if same_person(latest["by"], ticket.get("assignee")):
            problems.append("最新的 review 是 assignee 本人做的，需要其他人 review")
        if latest.get("open_critical") != 0:
            problems.append(f"最新的 review 仍有 {latest.get('open_critical')} 個未解的 🔴")
    return problems


def transition(
    repo: Path,
    ticket_id: str,
    to: str,
    who: Identity,
    now: str,
    *,
    note: str = "",
    blocked_reason: str | None = None,
) -> dict:
    ticket = load_ticket(repo, ticket_id)
    current = ticket["status"]
    if to not in STATUSES:
        raise TicketError(f"未知狀態：{to}")
    if to == current:
        raise TicketError(f"{ticket_id} 已經是 {to}")

    if to == "blocked":
        if current == "done":
            raise TicketError(f"{ticket_id} 已經 done，不能轉成 blocked")
        if not blocked_reason:
            raise TicketError("轉成 blocked 必須提供 blocked_reason（--reason）")
        ticket["blocked_reason"] = blocked_reason
    else:
        if current == "blocked":
            resume = _status_before_block(ticket)
            if to != resume:
                raise TicketError(f"{ticket_id} 解除 blocked 只能回到原狀態 {resume}")
        elif to not in _ALLOWED[current]:
            raise TicketError(f"不允許 {current} → {to}")
        problems: list[str] = []
        if to == "active":
            problems = _active_problems(repo, ticket, who)
        elif to == "verifying" and current == "active":
            problems = _verifying_problems(repo, ticket)
        elif to == "done":
            problems = done_problems(ticket)
        if problems:
            raise TicketError(f"{ticket_id} 無法轉成 {to}：\n- " + "\n- ".join(problems))
        if to == "active" and current != "blocked":
            ticket["assignee"] = who.as_ref()
        ticket["blocked_reason"] = None

    ticket["status"] = to
    ticket["history"].append({"by": who.as_ref(), "at": now, "from": current, "to": to, "note": note})
    save_ticket(repo, ticket)
    return ticket


def add_evidence(repo: Path, ticket_id: str, who: Identity, now: str, kind: str, fields: dict) -> dict:
    spec = EVIDENCE_FIELDS.get(kind)
    if spec is None:
        raise TicketError(f"evidence 種類必須是 {'/'.join(EVIDENCE_KINDS)} 之一，收到 {kind!r}")
    unknown = sorted(set(fields) - set(spec))
    if unknown:
        raise TicketError(f"{kind} evidence 不認得的欄位：{', '.join(unknown)}")
    for key, expected in spec.items():
        if not _matches(fields.get(key), expected):
            raise TicketError(f"{kind} evidence 缺少或型別錯誤：{key}（需要 {expected.__name__}）")
    ticket = load_ticket(repo, ticket_id)
    ticket["evidence"].append({"kind": kind, "by": who.as_ref(), "at": now, **fields})
    save_ticket(repo, ticket)
    return ticket


def add_dod(repo: Path, ticket_id: str, item: str) -> dict:
    ticket = load_ticket(repo, ticket_id)
    if item not in ticket["dod_pending"]:
        ticket["dod_pending"].append(item)
        save_ticket(repo, ticket)
    return ticket


def resolve_dod(repo: Path, ticket_id: str, item: str, who: Identity, now: str) -> dict:
    ticket = load_ticket(repo, ticket_id)
    if item not in ticket["dod_pending"]:
        raise TicketError(f"{item!r} 不在 {ticket_id} 的 dod_pending：{ticket['dod_pending']}")
    ticket["dod_pending"].remove(item)
    ticket["evidence"].append({"kind": "note", "by": who.as_ref(), "at": now, "summary": f"完成 dod：{item}"})
    save_ticket(repo, ticket)
    return ticket
```

```python
def _origin_key(ticket: dict) -> tuple:
    created_by = ticket.get("created_by") or {}
    return (str(created_by.get("email", "")).lower(), ticket.get("created_at"))


def collisions(repo: Path, main_ref: str) -> list[str]:
    """本地票檔與 main_ref 上同號、但 created_by/created_at 不同的票號。"""
    if not gitutil.ref_exists(repo, main_ref):
        return []
    found = []
    for path in sorted((Path(repo) / TICKETS_DIR).glob("FW-*.json")):
        remote_text = gitutil.show_at(repo, main_ref, f"{TICKETS_DIR.as_posix()}/{path.name}")
        if remote_text is None:
            continue
        try:
            remote = json.loads(remote_text)
        except json.JSONDecodeError:
            continue
        local = jsonio.read_json(path)
        if _origin_key(local) != _origin_key(remote):
            found.append(path.stem)
    return found


def renumber(repo: Path, ticket_id: str, who: Identity, now: str, main_ref: str) -> str:
    ticket = load_ticket(repo, ticket_id)
    old_path = ticket_path(repo, ticket_id)
    new_id, _ = allocate_id(repo, main_ref)
    ticket["id"] = new_id
    ticket["history"].append({
        "by": who.as_ref(), "at": now, "from": ticket["status"], "to": ticket["status"],
        "note": f"由 {ticket_id} 改號為 {new_id}（撞號）",
    })
    save_ticket(repo, ticket)
    old_path.unlink()
    return new_id


def check_all(repo: Path) -> list[str]:
    errors: list[str] = []
    active_by_person: dict[str, list[str]] = {}
    directory = Path(repo) / TICKETS_DIR
    paths = sorted(directory.glob("*.json")) if directory.is_dir() else []
    for path in paths:
        try:
            ticket = jsonio.read_json(path)
        except ValueError as exc:
            errors.append(f"{path.name}：JSON 無法解析（{exc}）")
            continue
        problems = validate_ticket(ticket, path.name)
        errors.extend(problems)
        if problems:
            continue
        if ticket["status"] == "active":
            active_by_person.setdefault(ticket["assignee"]["email"].lower(), []).append(ticket["id"])
        if ticket["status"] == "done":
            errors.extend(f"{ticket['id']}（done）：{p}" for p in done_problems(ticket))
    for email, ids in sorted(active_by_person.items()):
        if len(ids) > 1:
            errors.append(f"{email} 同時有多張 active：{', '.join(ids)}（每人一次一票）")
    return errors
```

```python
"""產生 feature_list.json 索引。commit evidence 由 git log 動態算出，不寫入票檔（spec §4.3）。"""
from __future__ import annotations

import re
from collections import Counter
from pathlib import Path

from . import gitutil, jsonio, tickets

TICKET_TOKEN = re.compile(r"\bFW-\d{4,}\b")
INDEX_FILE = "feature_list.json"
_FIELD = "\x1f"
_RECORD = "\x1e"


def commits_by_ticket(repo: Path) -> dict[str, list[dict]]:
    if not gitutil.has_commits(repo):
        return {}
    out = gitutil.git(repo, "log", f"--format=%h{_FIELD}%an{_FIELD}%aI{_FIELD}%s{_FIELD}%b{_RECORD}")
    result: dict[str, list[dict]] = {}
    for record in out.split(_RECORD):
        record = record.strip("\n")
        if not record:
            continue
        sha, author, at, subject, body = (record.split(_FIELD) + [""] * 5)[:5]
        entry = {"commit": sha, "author": author, "at": at, "subject": subject}
        for ticket_id in sorted(set(TICKET_TOKEN.findall(f"{subject}\n{body}"))):
            result.setdefault(ticket_id, []).append(entry)
    return result


def build_index(repo: Path, now: str) -> dict:
    commits = commits_by_ticket(repo)
    rows = []
    invalid = []
    directory = Path(repo) / tickets.TICKETS_DIR
    for path in sorted(directory.glob("FW-*.json")) if directory.is_dir() else []:
        try:
            ticket = jsonio.read_json(path)
        except ValueError:
            invalid.append(path.name)
            continue
        if tickets.validate_ticket(ticket, path.name):
            invalid.append(path.name)
            continue
        rows.append({
            "id": ticket["id"],
            "title": ticket["title"],
            "status": ticket["status"],
            "priority": ticket["priority"],
            "area": ticket["area"],
            "assignee": (ticket["assignee"] or {}).get("name"),
            "requires_hil": ticket["requires_hil"],
            "dod_pending": ticket["dod_pending"],
            "evidence_counts": dict(Counter(item["kind"] for item in ticket["evidence"])),
            "commits": commits.get(ticket["id"], []),
        })
    rows.sort(key=lambda row: (row["priority"], row["id"]))
    return {
        "_generated": "由 harness/scripts/build_index.py 產生，勿手動編輯；此檔不進 git。票檔在 harness/tickets/。",
        "generated_at": now,
        "status_legend": tickets.STATUS_LEGEND,
        "tickets": rows,
        "invalid": invalid,
    }


def write_index(repo: Path, now: str) -> Path:
    path = Path(repo) / INDEX_FILE
    jsonio.write_json_atomic(path, build_index(repo, now))
    return path
```

```python
#!/usr/bin/env python3
"""票檔操作的唯一入口（spec §5）。所有寫入都自動帶上 git 身分。

  new       --title T --area A [--priority N] [--behavior B] [--step S]... [--no-hil]
  claim     FW-0001
  status    FW-0001 <狀態> [--note N] [--reason R]     （done 需在互動式終端機由人確認）
  evidence  FW-0001 check --commit SHA --summary S --passed|--failed
  evidence  FW-0001 review --ref PATH --open-critical N
  evidence  FW-0001 hil --ref PATH --summary S
  evidence  FW-0001 note --summary S
  dod       add|done FW-0001 "項目"
  list      [--mine] [--status S]
  show      FW-0001
  collisions
  renumber  FW-0001
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))

from harnesslib import cli, gitutil, tickets  # noqa: E402
from harnesslib.identity import IdentityError, current_identity  # noqa: E402

WRITE_COMMANDS = {"new", "claim", "status", "evidence", "dod", "renumber"}


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="ticket.py", description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    sub = parser.add_subparsers(dest="cmd", required=True)

    new = sub.add_parser("new")
    new.add_argument("--title", required=True)
    new.add_argument("--area", required=True)
    new.add_argument("--priority", type=int, default=3)
    new.add_argument("--behavior", default="")
    new.add_argument("--step", action="append", default=[])
    new.add_argument("--no-hil", action="store_true")

    claim = sub.add_parser("claim")
    claim.add_argument("id")

    status = sub.add_parser("status")
    status.add_argument("id")
    status.add_argument("to", choices=tickets.STATUSES)
    status.add_argument("--note", default="")
    status.add_argument("--reason")

    evidence = sub.add_parser("evidence")
    evidence.add_argument("id")
    evidence.add_argument("kind", choices=tickets.EVIDENCE_KINDS)
    evidence.add_argument("--commit")
    evidence.add_argument("--summary")
    evidence.add_argument("--ref")
    evidence.add_argument("--open-critical", type=int)
    outcome = evidence.add_mutually_exclusive_group()
    outcome.add_argument("--passed", dest="passed", action="store_true")
    outcome.add_argument("--failed", dest="passed", action="store_false")
    evidence.set_defaults(passed=None)

    dod = sub.add_parser("dod")
    dod.add_argument("action", choices=("add", "done"))
    dod.add_argument("id")
    dod.add_argument("item")

    listing = sub.add_parser("list")
    listing.add_argument("--mine", action="store_true")
    listing.add_argument("--status", choices=tickets.STATUSES)

    show = sub.add_parser("show")
    show.add_argument("id")

    sub.add_parser("collisions")

    renumber = sub.add_parser("renumber")
    renumber.add_argument("id")
    return parser


def _fetch(repo: Path) -> str:
    ref = cli.main_ref(repo)
    if not gitutil.fetch(repo, ref):
        print(f"[harness] 警告：無法 fetch {ref}，改用本地資料。", file=sys.stderr)
    return ref


def _confirm_done(ticket_id: str) -> bool:
    if not sys.stdin.isatty():
        print("[harness] 轉成 done 必須由人在互動式終端機執行（agent 不可自行標記 done）。", file=sys.stderr)
        return False
    answer = input(f"確認 {ticket_id} 已完成上板驗證並經他人 review？請輸入票號確認：")
    return answer.strip() == ticket_id


def _run(args: argparse.Namespace, repo: Path) -> int:
    needs_identity = args.cmd in WRITE_COMMANDS or getattr(args, "mine", False)
    who = current_identity(repo) if needs_identity else None

    if args.cmd == "new":
        ref = _fetch(repo)
        ticket, warning = tickets.create_ticket(
            repo, who, title=args.title, area=args.area, now=cli.now(), main_ref=ref,
            priority=args.priority, user_visible_behavior=args.behavior,
            verification_steps=args.step, requires_hil=not args.no_hil,
        )
        if warning:
            print(f"[harness] 警告：{warning}", file=sys.stderr)
        print(f"已開票 {ticket['id']}：{ticket['title']}（created_by {who.name}）")
        print("提醒：開票 commit 合入 main 之後，票號才算佔住。")
    elif args.cmd == "claim":
        ticket = tickets.transition(repo, args.id, "active", who, cli.now(), note="claim")
        print(f"{ticket['id']} 已由 {who.name} 認領（active）")
    elif args.cmd == "status":
        if args.to == "done" and not _confirm_done(args.id):
            return 1
        ticket = tickets.transition(repo, args.id, args.to, who, cli.now(), note=args.note, blocked_reason=args.reason)
        print(f"{ticket['id']}：{ticket['history'][-1]['from']} → {ticket['status']}")
    elif args.cmd == "evidence":
        candidates = {
            "commit": args.commit, "summary": args.summary, "ref": args.ref,
            "open_critical": args.open_critical, "passed": args.passed,
        }
        fields = {key: value for key, value in candidates.items() if value is not None}
        tickets.add_evidence(repo, args.id, who, cli.now(), args.kind, fields)
        print(f"{args.id} 已加入 {args.kind} evidence（by {who.name}）")
    elif args.cmd == "dod":
        if args.action == "add":
            tickets.add_dod(repo, args.id, args.item)
            print(f"{args.id} dod_pending 加入：{args.item}")
        else:
            tickets.resolve_dod(repo, args.id, args.item, who, cli.now())
            print(f"{args.id} dod_pending 完成：{args.item}")
    elif args.cmd == "list":
        for ticket in tickets.load_all(repo):
            if args.status and ticket.get("status") != args.status:
                continue
            assignee = ticket.get("assignee") or {}
            if args.mine and assignee.get("email", "").lower() != who.email.lower():
                continue
            print(f"{ticket['id']}  {ticket['status']:<9}  P{ticket['priority']}  {assignee.get('name', '-'):<12}  {ticket['title']}")
    elif args.cmd == "show":
        print(json.dumps(tickets.load_ticket(repo, args.id), ensure_ascii=False, indent=2))
    elif args.cmd == "collisions":
        found = tickets.collisions(repo, _fetch(repo))
        if not found:
            print("沒有撞號")
            return 0
        for ticket_id in found:
            print(f"撞號：{ticket_id}（執行 ticket.py renumber {ticket_id}）")
        return 1
    elif args.cmd == "renumber":
        new_id = tickets.renumber(repo, args.id, who, cli.now(), _fetch(repo))
        print(f"{args.id} → {new_id}。請修改分支上引用 {args.id} 的 commit 訊息。")
    return 0


def main(argv: list[str] | None = None) -> int:
    cli.setup_io()
    args = build_parser().parse_args(argv)
    try:
        return _run(args, cli.repo_root())
    except (tickets.TicketError, IdentityError, gitutil.GitError, ValueError) as exc:
        return cli.fail(str(exc))


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""commit 訊息規則：必須引用存在的票號，或屬於 harness:/chore: 等例外（spec §4.3）。"""
from __future__ import annotations

import re
from pathlib import Path

from . import tickets
from .index import TICKET_TOKEN

_EXEMPT_PREFIX = re.compile(r"^(harness|chore)(\([^)]*\))?!?:")
_GIT_GENERATED = ("Merge ", "Revert ", "fixup! ", "squash! ", "amend! ")


def check_message(repo: Path, text: str) -> list[str]:
    lines = [line for line in text.splitlines() if not line.startswith("#")]
    message = "\n".join(lines).strip()
    if not message:
        return []
    first = message.splitlines()[0]
    if first.startswith(_GIT_GENERATED) or _EXEMPT_PREFIX.match(first):
        return []
    ids = sorted(set(TICKET_TOKEN.findall(message)))
    if not ids:
        return ["commit 訊息必須包含票號（例如 FW-0012），或以 harness: / chore: 開頭"]
    missing = [ticket_id for ticket_id in ids if not tickets.ticket_path(repo, ticket_id).exists()]
    if missing:
        return [f"票號不存在：{', '.join(missing)}（先用 ticket.py new 開票）"]
    return []
```

```python
#!/usr/bin/env python3
"""commit-msg hook 呼叫：檢查 commit 訊息是否引用存在的票號。"""
from __future__ import annotations

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))

from harnesslib import cli, gitutil  # noqa: E402
from harnesslib.commitmsg import check_message  # noqa: E402


def main(argv: list[str] | None = None) -> int:
    cli.setup_io()
    args = sys.argv[1:] if argv is None else argv
    if len(args) != 1:
        return cli.fail("用法：commit_msg_check.py <commit 訊息檔>")
    try:
        repo = cli.repo_root()
    except gitutil.GitError as exc:
        return cli.fail(str(exc))
    text = Path(args[0]).read_text(encoding="utf-8", errors="replace")
    errors = check_message(repo, text)
    for error in errors:
        print(f"[harness] {error}", file=sys.stderr)
    return 1 if errors else 0


if __name__ == "__main__":
    sys.exit(main())
```
